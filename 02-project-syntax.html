├─ src/

CMake 在做什麼（builder/CMakeLists.txt）

  • CMake 不是編譯器，是「產生專案檔」的工具：讀 CMakeLists.txt → 產生 VS solution / Xcode 專案 / Makefile / Ninja。
  • Template 的 CMakeLists 大致做四件事：
    1. 設定 C++ 標準等編譯選項
    2. 有 CMakeLists 的 library（GLFW、Assimp…）用 add_subdirectory 一起編
    3. 把 src/ 的檔案編成執行檔，target_link_libraries 連結 library
    4. Header-only / 沒有 CMakeLists 的 library（GLM、stb…）手動 include_directories；並設定路徑方便 IDE 找資源
  • 新增 .cpp 檔：若 CMakeLists 用 file(GLOB ...) 收集 src，新增檔後要重新 Configure 才會被加入。
範例：從原始碼到執行檔的三個階段
階段做什麼Windows（CMake GUI + VS）Linux（CLI）
Configure讀 CMakeLists、找編譯器、檢查 library 能不能編按 Configurecmake -S builder -B build
Generate產生 IDE 專案檔 / Makefile按 Generate → 得到 .sln
Build真正呼叫編譯器把 .cpp 編成 exeVS 裡按 ▶ / Ctrl+Shift+Bcmake --build build

日常開發只要改 src/ 的程式 → 重新 Build 即可，不用每次跑 CMake。只有「新增 / 刪除檔案」或「改 CMakeLists」才要重新 Configure + Generate。

例：CMakeLists 中常見的寫法與意思

cmake_minimum_required(VERSION 3.16)
project(GPA)
set(CMAKE_CXX_STANDARD 17)                     # 用 C++17
add_subdirectory(glfw)                         # 編 GLFW（它有自己的 CMakeLists）
file(GLOB SRC ../src/*.cpp)                    # 收集 src 下所有 .cpp
add_executable(${PROJECT_NAME} ${SRC})         # 產生執行檔
target_include_directories(${PROJECT_NAME} PRIVATE glm)   # header-only：只要告訴編譯器路徑
target_link_libraries(${PROJECT_NAME} glfw)    # 連結 GLFW

程式骨架（GLFW + GLAD）

  • GLAD：OpenGL function loader，讓你能呼叫 3.3 / 4.x 的函式（沒載入就呼叫 → 直接 crash）。
  • GLFW：建立視窗、OpenGL context、處理輸入。
  • 順序重要：GLFW init → hint → 建視窗 → glfwMakeContextCurrent → 才能 init GLAD → 之後才能呼叫任何 gl*。
#include <glad/glad.h>   // 一定要在 GLFW 之前 include
#include <GLFW/glfw3.h>

int main() {
    // 1. Init GLFW + 指定 OpenGL 版本（依 template 設定，macOS 最高 4.1）
    glfwInit();
    glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, 4);
    glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, 1);
    glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);
#ifdef __APPLE__
    glfwWindowHint(GLFW_OPENGL_FORWARD_COMPAT, GL_TRUE);
#endif
    GLFWwindow* window = glfwCreateWindow(800, 600, "GPA", nullptr, nullptr);
    glfwMakeContextCurrent(window);

    // 2. Callbacks
    glfwSetFramebufferSizeCallback(window, framebuffer_size_callback);
    glfwSetKeyCallback(window, key_callback);

    // 3. Init GLAD（GLAD1 寫法；GLAD2 為 gladLoadGL(glfwGetProcAddress)）
    gladLoadGLLoader((GLADloadproc)glfwGetProcAddress);

    // 4. Setup：shader program、VAO/VBO、texture…（只做一次）
    glEnable(GL_DEPTH_TEST);

    double last = glfwGetTime();
    while (!glfwWindowShouldClose(window)) {
        double now = glfwGetTime(), dt = now - last; last = now;
        update(dt);                                          // 用 deltaTime 做動畫，速度不受 FPS 影響
        glClearColor(0.1f, 0.1f, 0.1f, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
        render();                                            // glUseProgram → bind VAO → glUniform → glDraw*
        glfwSwapBuffers(window);
        glfwPollEvents();
    }
    // 5. Cleanup
    glDeleteVertexArrays(1, &vao); glDeleteProgram(program);
    glfwTerminate();
}

把 shader 寫成字串以便單檔閱讀；作業中通常放在 shaders/*.glsl 再讀檔。

#include <glad/glad.h>
#include <GLFW/glfw3.h>
#include <cstdio>

const char* vsSrc = R"(
#version 410 core
layout(location = 0) in vec2 aPos;     // attrib 0：位置
layout(location = 1) in vec3 aColor;   // attrib 1：顏色
out vec3 vColor;                       // 傳給 fragment shader
void main() {
    vColor = aColor;
    gl_Position = vec4(aPos, 0.0, 1.0);
})";
const char* fsSrc = R"(
#version 410 core
in vec3 vColor;                        // 名稱、型別要和 VS 的 out 一樣
out vec4 fragColor;
void main() { fragColor = vec4(vColor, 1.0); })";

GLuint compile(GLenum type, const char* src) {
    GLuint s = glCreateShader(type);
    glShaderSource(s, 1, &src, NULL);
    glCompileShader(s);
    GLint ok; glGetShaderiv(s, GL_COMPILE_STATUS, &ok);
    if (!ok) { char log[1024]; glGetShaderInfoLog(s, 1024, NULL, log); printf("%s\n", log); }
    return s;
}

int main() {
    glfwInit();
    glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, 4);
    glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, 1);
    glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);
    GLFWwindow* win = glfwCreateWindow(800, 600, "Triangle", NULL, NULL);
    glfwMakeContextCurrent(win);
    gladLoadGLLoader((GLADloadproc)glfwGetProcAddress);

    // --- shader program ---
    GLuint vs = compile(GL_VERTEX_SHADER, vsSrc), fs = compile(GL_FRAGMENT_SHADER, fsSrc);
    GLuint prog = glCreateProgram();
    glAttachShader(prog, vs); glAttachShader(prog, fs); glLinkProgram(prog);
    glDeleteShader(vs); glDeleteShader(fs);

    // --- 頂點資料：x, y, r, g, b ---
    float verts[] = {
        -0.5f, -0.5f,   1, 0, 0,    // 左下 紅
         0.5f, -0.5f,   0, 1, 0,    // 右下 綠
         0.0f,  0.5f,   0, 0, 1,    // 上   藍
    };
    GLuint vao, vbo;
    glGenVertexArrays(1, &vao); glBindVertexArray(vao);
    glGenBuffers(1, &vbo); glBindBuffer(GL_ARRAY_BUFFER, vbo);
    glBufferData(GL_ARRAY_BUFFER, sizeof(verts), verts, GL_STATIC_DRAW);
    glVertexAttribPointer(0, 2, GL_FLOAT, GL_FALSE, 5 * sizeof(float), (void*)0);
    glEnableVertexAttribArray(0);
    glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE, 5 * sizeof(float), (void*)(2 * sizeof(float)));
    glEnableVertexAttribArray(1);

    while (!glfwWindowShouldClose(win)) {
        glClearColor(0.1f, 0.1f, 0.1f, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT);
        glUseProgram(prog);
        glBindVertexArray(vao);
        glDrawArrays(GL_TRIANGLES, 0, 3);
        glfwSwapBuffers(win);
        glfwPollEvents();
    }
    glDeleteBuffers(1, &vbo); glDeleteVertexArrays(1, &vao); glDeleteProgram(prog);
    glfwTerminate();
}

Callbacks

Callback 是什麼？

你寫一個函式，交給 GLFW 保管；等事件發生（按鍵、移動滑鼠），GLFW 在 glfwPollEvents() 裡幫你呼叫它。你不會自己呼叫 callback。

bool wireframe = false;                         // callback 要改的狀態放全域（或用 glfwSetWindowUserPointer）

void key_callback(GLFWwindow* w, int key, int scancode, int action, int mods) {
    if (key == GLFW_KEY_F && action == GLFW_PRESS) wireframe = !wireframe;
}

int main() {
    ...
    glfwSetKeyCallback(window, key_callback);   // 註冊：只傳函式名，不加 ()
    while (...) {
        glPolygonMode(GL_FRONT_AND_BACK, wireframe ? GL_LINE : GL_FILL);
        ...
        glfwPollEvents();                       // 按了 F → key_callback 在這裡被呼叫
    }
}

函式的參數型別必須和 GLFW 規定的簽名完全一樣（見下表），否則編譯錯誤。

用途註冊Callback 簽名 / 備註
鍵盤glfwSetKeyCallbackvoid(GLFWwindow*, int key, int scancode, int action, int mods)
例：key==GLFW_KEY_E && action==GLFW_PRESS；組合鍵 mods & GLFW_MOD_CONTROL
滑鼠位置glfwSetCursorPosCallbackvoid(GLFWwindow*, double x, double y)；或每幀 glfwGetCursorPos 主動查
滑鼠按鍵 / 滾輪glfwSetMouseButtonCallback / glfwSetScrollCallback做相機操作常用
視窗大小glfwSetFramebufferSizeCallbackvoid(GLFWwindow*, int w, int h) → glViewport(0,0,w,h)，重算 projection aspect；有自訂 FBO 要重建。HW2 / Final 會用到
持續按住移動相機：用 glfwGetKey(window, GLFW_KEY_W) == GLFW_PRESS 在 update 中每幀查詢，比 key callback 的 PRESS/REPEAT 平順。

Shader Program 建立流程

GLuint compile(GLenum type, const char* src) {
    GLuint s = glCreateShader(type);           // GL_VERTEX_SHADER / GL_FRAGMENT_SHADER
    glShaderSource(s, 1, &src, NULL);
    glCompileShader(s);
    GLint ok; glGetShaderiv(s, GL_COMPILE_STATUS, &ok);
    if (!ok) { char log[1024]; glGetShaderInfoLog(s, 1024, NULL, log); printf("%s\n", log); }  // 一定要印！
    return s;
}
GLuint program = glCreateProgram();
GLuint vs = compile(GL_VERTEX_SHADER, vsSrc), fs = compile(GL_FRAGMENT_SHADER, fsSrc);
glAttachShader(program, vs); glAttachShader(program, fs);
glLinkProgram(program);                        // 檢查 GL_LINK_STATUS / glGetProgramInfoLog
glDeleteShader(vs); glDeleteShader(fs);        // link 完 shader object 可刪
// render 時： glUseProgram(program);
  • Compile 很吃 CPU → 所有 shader 在啟動時載入，不要每幀 compile。
  • Shader 寫在 shaders/*.glsl，C++ 讀檔成字串再丟 glShaderSource；執行時的工作目錄錯了會讀不到檔（黑畫面）。
  • 一個 shader object 可被多個 program 共用（in/out 介面要相符）。

OpenGL Objects：資料怎麼進 Shader

四種最常用的資料通道。黃色 = 「binding point / unit」這個中介編號，最容易搞混。

GLuint vao, vbo;
glGenVertexArrays(1, &vao); glBindVertexArray(vao);      // 先 bind VAO，後面設定都記在它身上
glGenBuffers(1, &vbo);       glBindBuffer(GL_ARRAY_BUFFER, vbo);
glBufferData(GL_ARRAY_BUFFER, sizeof(verts), verts, GL_STATIC_DRAW);
// 每個 vertex: pos(3) + color(3)，stride = 6 floats
glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, 6*sizeof(float), (void*)0);
glEnableVertexAttribArray(0);
glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE, 6*sizeof(float), (void*)(3*sizeof(float)));
glEnableVertexAttribArray(1);
// index buffer（可選）：glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, ebo) 也會記在 VAO → glDrawElements
glBindVertexArray(0);
// render: glBindVertexArray(vao); glDrawArrays(GL_TRIANGLES, 0, 3);
白話解釋 VBO 與 VAO
  • VBO = GPU 上的一塊記憶體，裡面只有一長串數字，GPU 不知道哪些是位置、哪些是顏色。
  • VAO = 一張「讀法說明書」：attrib 0 從哪個 VBO、第幾個 byte 開始、每次讀幾個 float、讀完跳幾個 byte 到下一個頂點。

上面程式的 VBO 記憶體長這樣（每格 4 bytes 的 float）：

byte048121620242832364044…
內容x₀y₀z₀r₀g₀b₀x₁y₁z₁r₁g₁b₁…
  • attrib 0（位置）：size = 3、offset = 0、stride = 24（6 個 float × 4 bytes，跳到下一個頂點的距離）
  • attrib 1（顏色）：size = 3、offset = 12（跳過 x,y,z）、stride = 24
  • Shader 端 layout(location = 0) in vec3 aPos; 的 0 就對應 attrib 0。

好處：畫圖時只要 glBindVertexArray(vao) 一行就恢復所有設定；多個模型 = 多個 VAO，切換很快。

Buffer target用途
GL_ARRAY_BUFFERVertex buffer（VBO）
GL_ELEMENT_ARRAY_BUFFERIndex buffer（EBO）
GL_UNIFORM_BUFFERUniform block 資料（UBO）
GL_SHADER_STORAGE_BUFFERSSBO（4.3+，compute 用）
GL_COPY_READ/WRITE_BUFFER、GL_TEXTURE_BUFFER、GL_TRANSFORM_FEEDBACK_BUFFER複製、texture buffer、transform feedback

Usage hint（只是提示，不影響正確性）：STATIC 設一次用很多次｜DYNAMIC 常改常用｜STREAM 設一次用幾次；× DRAW（App→GPU）/ READ（GPU→App）/ COPY（GPU→GPU）。部分更新用 glBufferSubData 或 glMapBuffer/glUnmapBuffer。

GLint loc = glGetUniformLocation(program, "mvp");   // 回傳 -1 = 名字錯或被 compiler 優化掉（沒用到）
glUseProgram(program);                              // glUniform* 作用在「目前使用中的 program」
glUniformMatrix4fv(loc, 1, GL_FALSE, glm::value_ptr(mvp));  // GL_FALSE：GLM 已是 column-major，不用轉置
glUniform1f(glGetUniformLocation(program, "time"), t);
glUniform3fv(loc2, 1, &color[0]);
  • Uniform 在一次 draw call 內對所有頂點 / fragment 都是同一個值（相機、光源、時間）。
  • 值存在 program 內：換 program 要重設；location 可在 init 時查一次存起來。
  • VS 和 FS 宣告同名同型的 uniform 視為同一個。

Texture 要點

UV 座標是什麼？

貼圖不管實際多少 pixel，都用 (u, v) ∈ [0, 1] 表示位置：(0,0) 左下、(1,1) 右上、(0.5,0.5) 正中央。每個頂點帶一組 UV（從模型檔來），rasterizer 插值後，FS 用 texture(tex, uv) 取色。

// 一個貼滿整張圖的正方形：位置 (x,y) + UV (u,v)
float quad[] = {
    -1, -1,   0, 0,     // 左下 ↔ 圖片左下
     1, -1,   1, 0,     // 右下 ↔ 圖片右下
     1,  1,   1, 1,     // 右上
    -1,  1,   0, 1,     // 左上
};
// UV 改成 0~3 → 搭配 GL_REPEAT 會重複貼 3×3 次（地板磁磚常用）

流程整理：載入圖片（stb_image）→ 建立 texture 並上傳 → 設定 filter / wrap → 畫圖前 綁到 texture unit → shader 的 sampler2D 設成該 unit 編號 → FS 用 UV 取樣。

Filter（Magnification：貼圖被放大）
GL_NEARESTGL_LINEAR
  • NEAREST：最近 texel（像素風、Minecraft）
  • LINEAR：2×2 texel 雙線性插值
  • Minification（縮小）用 mipmap：每層是上一層 2×2 平均，GL_LINEAR_MIPMAP_LINEAR = trilinear；斜角更清楚用 anisotropic filtering
  • MIN_FILTER 預設帶 mipmap：沒產生 mipmap 又沒改 filter → texture incomplete → 取樣得黑色
Wrap mode（UV 超出 [0,1]）

示意 UV 範圍 [-1, 2]（中間格為 [0,1]）。設定：GL_TEXTURE_WRAP_S / _T。

  • GLSL 取樣：texture(sampler, uv) 走 filter + wrap；texelFetch(sampler, ivec2, lod) 直接用整數座標拿 texel、不過濾。
  • Texture target：GL_TEXTURE_2D（最常用）、GL_TEXTURE_CUBE_MAP（skybox / environment map）、GL_TEXTURE_2D_ARRAY、GL_TEXTURE_3D…對應 sampler 型別 sampler2D、samplerCube、sampler2DArray。
  • Depth texture 可設 GL_TEXTURE_COMPARE_MODE 做 shadow mapping（之後課程）。
  • 影像通常上下顛倒：OpenGL UV 原點在左下，多數圖檔在左上 → 載入時 flip（如 stbi_set_flip_vertically_on_load(true)）。

載入圖片：stb_image（Appendix - Image Loading）

  • 圖檔（png / jpg）是壓縮過的格式，OpenGL 看不懂 → 要先解碼成一格一格的 RGBA bytes（bitmap），再交給 glTexSubImage2D。
  • 解碼方式：自己照檔案格式規格寫（不實際）、單一格式 library（libpng、libjpeg）、影像處理 library（OpenCV）、或課程用的 stb_image。
  • stb_image.h：單一 header 檔、支援 png / jpg / tga / bmp / psd / gif / hdr / pic。同系列還有 stb_image_write.h（存圖，例如截圖）、stb_image_resize.h（縮放）。網站：stb.handmade.network
Header-only library 的規則：所有檔案都可以 #include "STB/stb_image.h"，但只有一個 .cpp 要在 include 前加 #define STB_IMAGE_IMPLEMENTATION（產生函式本體）。
沒加 → link error「undefined reference to stbi_load」；兩個 .cpp 都加 → link error「multiple definition」。
// texture.h —— 只放宣告
#include "STB/stb_image.h"
struct texture_data { int width; int height; unsigned char* data; };
texture_data load_Image(const char* path);

// main.cpp —— 唯一一處放實作
#define STB_IMAGE_IMPLEMENTATION
#include "STB/stb_image.h"

texture_data load_Image(const char* path) {
    texture_data tex = {};
    int n;                                                  // 原圖的通道數（RGB=3、RGBA=4）
    stbi_set_flip_vertically_on_load(true);                 // 上下翻轉，配合 OpenGL UV 原點在左下
    stbi_uc* data = stbi_load(path, &tex.width, &tex.height, &n, 4);  // 最後的 4 = 強制轉成 RGBA
    if (data != NULL) {
        size_t size = tex.width * tex.height * 4;           // 每個 pixel 4 bytes
        tex.data = new unsigned char[size];
        memcpy(tex.data, data, size);
        stbi_image_free(data);                              // stb 配置的記憶體要用 stb 的函式釋放
    } else {
        printf("load %s failed: %s\n", path, stbi_failure_reason());   // 最常見：路徑錯
    }
    return tex;                                             // 用完記得 delete[] tex.data
}

// 上傳到 OpenGL
texture_data img = load_Image("assets/wood.png");
glBindTexture(GL_TEXTURE_2D, tex);
glTexStorage2D(GL_TEXTURE_2D, 1, GL_RGBA8, img.width, img.height);
glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, img.width, img.height, GL_RGBA, GL_UNSIGNED_BYTE, img.data);
delete[] img.data;

強制 4 通道（最後參數填 4）的好處：不管原圖是 RGB 還是 RGBA，上傳時都用 GL_RGBA，也不會遇到 RGB 每列沒對齊 4 bytes 的 glPixelStorei 問題。

GLSL 速查

// vertex shader
#version 410 core
layout(location = 0) in vec3 aPos;
layout(location = 1) in vec3 aColor;
uniform mat4 mvp;
out VS_OUT { vec3 color; } vs_out;   // interface block
void main() {
    vs_out.color = aColor;
    gl_Position = mvp * vec4(aPos, 1.0);
}
// fragment shader
#version 410 core
in VS_OUT { vec3 color; } fs_in;     // block 名稱要相同
uniform sampler2D tex;
out vec4 fragColor;
void main() {
    fragColor = vec4(fs_in.color, 1.0);
}
讀上面的程式
  • #version 410 core：第一行，宣告 GLSL 版本（4.1、core profile）。少了或打錯就無法編譯。
  • layout(location = 0) in vec3 aPos;：從 VAO 的 attrib 0 讀進來的 vec3，每個頂點不同。
  • out VS_OUT { vec3 color; } vs_out;：要交給下一個 stage 的資料。FS 必須宣告同名的 in block（block 名 VS_OUT 要相同，instance 名 vs_out / fs_in 可以不同）。不用 block 時，out vec3 vColor; ↔ in vec3 vColor; 名稱要完全一樣。
  • gl_Position：內建變數，不用宣告，VS 一定要寫入。
  • out vec4 fragColor;：FS 的輸出，名字自取，就是這個像素的顏色 (r,g,b,a)，範圍 0~1。

Swizzle 範例：vec4 c = vec4(0.1, 0.2, 0.3, 1.0); → c.rgb = (0.1,0.2,0.3)、c.bgr = (0.3,0.2,0.1)、c.xx = (0.1,0.1)、vec4(c.rgb * 2.0, c.a) 只把顏色加倍、alpha 不動。

項目重點
型別float int uint bool；vec2/3/4、ivec*、uvec*、bvec*；mat2/3/4（column-major）；sampler2D…
Constructor✓ vec3(1.0)、vec4(v3, 1.0)、vec3(v4)｜✗ vec2 a = (1.0, 1.0);、✗ vec4 b(1,2,3,4);
Swizzle.xyzw = .rgba = .stpq；可重複/重排 v.zyx、v.rrrr；不能混用 v.xyba ✗
運算* 對 vector 是逐元素；mat * vec 是矩陣乘；內積用 dot()。內建 normalize cross dot length mix clamp max pow reflect transpose inverse
長度向量長度 length(v)；陣列/矩陣元素數 m.length()
型別嚴格不會自動轉型：float x = 1; 在舊版會錯，寫 1.0；gl_FragCoord - vec2(1) ✗ → gl_FragCoord.xy - vec2(1)
陣列 / structfloat a[3] = float[3](1.,2.,3.);；支援 array of arrays、struct（同 C）
Qualifierin / out（stage 間傳遞，同名相連）、uniform、layout(location=N)、flat（不插值）
Built-inVS：gl_Position（out）、gl_VertexID、gl_InstanceID；FS：gl_FragCoord（視窗座標 x,y,z,1/w）
不要從投影片複製 code：會混入非 ASCII 字元（全形引號、減號 –）→ shader compile error。看 console 的錯誤行號。

  • Header-only，語法仿 GLSL；#include <glm/glm.hpp> + <glm/gtc/matrix_transform.hpp> + <glm/gtc/type_ptr.hpp>
  • 要 swizzle（v.xyz()）需在 include 前 #define GLM_FORCE_SWIZZLE；GTX 擴充需 #define GLM_ENABLE_EXPERIMENTAL
using namespace glm;                        // 可選，程式較短
mat4 I(1.0f);                               // identity（新版 GLM mat4 m; 不保證是 identity，請明寫 1.0f）
mat4 M = translate(I, vec3(tx, ty, tz));
M = rotate(M, radians(45.0f), vec3(0, 1, 0));  // 角度用弧度！
M = scale(M, vec3(2.0f));                   // 結果 M = T * R * S：「後寫的先作用在頂點上」
mat4 V = lookAt(vec3(0, 2, 5) /*eye*/, vec3(0) /*center*/, vec3(0, 1, 0) /*up*/);
mat4 P = perspective(radians(45.0f), (float)w / h, 0.1f, 100.0f);

mat4 mvp = P * V * M;

範例：讓立方體在 (2,0,0) 原地自轉，每秒 90° 
float t = (float)glfwGetTime();
glm::mat4 model(1.0f);
model = glm::translate(model, glm::vec3(2, 0, 0));                  // ③ 最後：搬到 (2,0,0)
model = glm::rotate(model, glm::radians(90.0f * t), glm::vec3(0,1,0)); // ② 再：繞自己的 y 軸轉
model = glm::scale(model, glm::vec3(0.5f));                         // ① 先：縮成一半
// 程式由上往下寫 ③②①，實際作用在頂點上的順序是 ①②③

glm::mat4 view = glm::lookAt(glm::vec3(0, 3, 8), glm::vec3(0), glm::vec3(0, 1, 0));
glm::mat4 proj = glm::perspective(glm::radians(45.0f), 800.0f / 600.0f, 0.1f, 100.0f);
glm::mat4 mvp  = proj * view * model;
glUseProgram(prog);
glUniformMatrix4fv(glGetUniformLocation(prog, "mvp"), 1, GL_FALSE, glm::value_ptr(mvp));

若把 translate 和 rotate 兩行對調，立方體會變成繞世界原點公轉（半徑 2）。

translate(M, t) 回傳 M * T（右乘）→ 對頂點來說是「先做 t、再做原本的 M」。所以 code 由上往下寫 T、R、S，實際作用順序是 S → R → T。

常見坑（黑畫面檢查表）

  1. Shader compile / link 有錯但沒印 log
  2. 沒 bind VAO 就 draw（core profile 必須有 VAO）
  3. glUniform* 前沒 glUseProgram；location = -1
  4. glVertexAttribPointer 的 stride / offset 以 byte 計，算錯
  5. 開了 depth test 但沒 clear depth buffer；或 near/far 設錯讓物體在視錐外
  6. 矩陣乘法順序寫反；rotate 傳了角度不是弧度
  7. 相機看錯方向（OpenGL 相機看 -z），物體在背後
  8. Texture 沒 mipmap 但 MIN_FILTER 用 mipmap 模式；sampler 設成 texture ID 而非 unit 編號
  9. 檔案路徑：執行時工作目錄不是專案根 → assets / shaders 讀不到
  10. Face culling 開著但模型頂點順序是 CW

Debug 好用：glGetError()、glPolygonMode(GL_FRONT_AND_BACK, GL_LINE) 看線框、FS 直接輸出常數色或 normal 檢查管線。
